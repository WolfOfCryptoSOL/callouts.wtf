WebSocket API

Receive Pump.fun callouts in your own code the moment we see them. Available from the Builder plan; "every callout" mode from Pro.

1. Get a key

In @Calloutswtfbot: 📬 Deliver To → 🔑 API key. The key is shown once — store it safely. Creating a new key beyond your plan's limit retires your oldest key.

2. Connect

wss://api.callouts.wtf/v1/stream
Authorization: Bearer <your key>

One live connection per key; a new connection with the same key replaces the old one. The server sends a heartbeat ping every 15 seconds.

On connect you receive:

{"type": "welcome", "plan": "pro", "all_callouts": true, "available": true}

3. Subscribe

By default you receive the callers you added in the bot (originals and updates). Send a subscribe message to change what you get:

{"type": "subscribe", "all": false, "kinds": ["original", "update", "repost", "quote"]}
FieldMeaning
alltrue = every Pump.fun callout (Pro and Fast Lane). false = only your callers.
kindsAny of original, update, repost, quote. Reposts and quotes need Builder or higher.

4. Callout events

{
  "type": "callout",
  "source": "pumpfun",
  "kind": "original",
  "callout_id": "8a0be443-e776-4f09-8fe3-d4eabccc60b7",
  "update_index": 0,
  "caller": {"handle": "EustazZ", "user_id": "6d1b989f-…", "wallet": "FqamE7xrah…"},
  "token": {"mint": "DdYmLRMQLtoMqutPCapAHPYsrnVZxgNZB2ZG4eTBpump", "chain": "solana"},
  "linked_callout_id": null,
  "pump_created_at": 1793268320808,
  "sent_at": 1793268320941,
  "fast": true
}
FieldMeaning
kindoriginal, update (the caller's own update on that callout), repost or quote.
update_index0 for the original; 1, 2, … for the caller's updates.
token.chainsolana, base, bsc, ethereum or robinhood.
linked_callout_idFor reposts and quotes: the callout they refer to.
pump_created_atPump.fun's own creation time (ms). sent_at - pump_created_at is our end-to-end time to you.
fasttrue when the callout came through the fast path.

Each callout is sent once. If the service is unavailable you receive {"type": "status", "source": "pumpfun", "available": false}, and true when it is back.

5. Catch-up (Pro and Fast Lane)

Turn on catch-up in the bot (⚙️ Settings), then reconnect with ?since=<ms timestamp> to receive callouts you missed in the last 10 minutes.

Examples

Set your key in the CALLOUTS_KEY environment variable, then run one of these. Each prints the caller, the token and our end-to-end time.

pip install websockets

import asyncio, json, os
import websockets

URL = "wss://api.callouts.wtf/v1/stream"
HEADERS = {"Authorization": "Bearer " + os.environ["CALLOUTS_KEY"]}

async def main():
    # Reconnects automatically after a dropped connection.
    async for ws in websockets.connect(URL, additional_headers=HEADERS):
        try:
            await ws.send(json.dumps({"type": "subscribe", "all": False, "kinds": ["original", "update"]}))
            async for raw in ws:
                e = json.loads(raw)
                if e["type"] == "callout":
                    print(e["caller"]["handle"], e["token"]["mint"], e["sent_at"] - e["pump_created_at"], "ms")
        except websockets.ConnectionClosed:
            continue

asyncio.run(main())

npm install ws (save as stream.mjs, run: node stream.mjs)

import WebSocket from "ws";

function connect() {
  const ws = new WebSocket("wss://api.callouts.wtf/v1/stream", {
    headers: { Authorization: `Bearer ${process.env.CALLOUTS_KEY}` },
  });
  ws.on("open", () => {
    ws.send(JSON.stringify({ type: "subscribe", all: false, kinds: ["original", "update"] }));
  });
  ws.on("message", (data) => {
    const e = JSON.parse(data);
    if (e.type === "callout") {
      console.log(e.caller.handle, e.token.mint, e.sent_at - e.pump_created_at, "ms");
    }
  });
  ws.on("close", () => setTimeout(connect, 1000)); // reconnect
  ws.on("error", () => {}); // "close" follows every error
}

connect();

import WebSocket from "ws";

interface Callout {
  type: "callout";
  source: "pumpfun";
  kind: "original" | "update" | "repost" | "quote";
  callout_id: string;
  update_index: number;
  caller: { handle: string | null; user_id: string | null; wallet: string };
  token: { mint: string; chain: "solana" | "base" | "bsc" | "ethereum" | "robinhood" };
  linked_callout_id: string | null;
  pump_created_at: number;
  sent_at: number;
  fast: boolean;
}

function connect(): void {
  const ws = new WebSocket("wss://api.callouts.wtf/v1/stream", {
    headers: { Authorization: `Bearer ${process.env.CALLOUTS_KEY}` },
  });
  ws.on("open", () => {
    ws.send(JSON.stringify({ type: "subscribe", all: false, kinds: ["original", "update"] }));
  });
  ws.on("message", (data) => {
    const e = JSON.parse(data.toString());
    if (e.type === "callout") {
      const c = e as Callout;
      console.log(c.caller.handle, c.token.mint, c.sent_at - c.pump_created_at, "ms");
    }
  });
  ws.on("close", () => setTimeout(connect, 1000));
  ws.on("error", () => {});
}

connect();

go get github.com/gorilla/websocket

package main

import (
	"log"
	"net/http"
	"os"
	"time"

	"github.com/gorilla/websocket"
)

type Event struct {
	Type   string `json:"type"`
	Kind   string `json:"kind"`
	Caller struct {
		Handle string `json:"handle"`
		Wallet string `json:"wallet"`
	} `json:"caller"`
	Token struct {
		Mint  string `json:"mint"`
		Chain string `json:"chain"`
	} `json:"token"`
	PumpCreatedAt int64 `json:"pump_created_at"`
	SentAt        int64 `json:"sent_at"`
}

func main() {
	header := http.Header{"Authorization": {"Bearer " + os.Getenv("CALLOUTS_KEY")}}
	for {
		conn, _, err := websocket.DefaultDialer.Dial("wss://api.callouts.wtf/v1/stream", header)
		if err != nil {
			log.Println("connect:", err)
			time.Sleep(time.Second)
			continue
		}
		conn.WriteJSON(map[string]any{"type": "subscribe", "all": false, "kinds": []string{"original", "update"}})
		for {
			var e Event
			if err := conn.ReadJSON(&e); err != nil {
				log.Println("read:", err)
				break
			}
			if e.Type == "callout" {
				log.Println(e.Caller.Handle, e.Token.Mint, e.SentAt-e.PumpCreatedAt, "ms")
			}
		}
		conn.Close()
		time.Sleep(time.Second)
	}
}

Cargo.toml: tokio = { version = "1", features = ["full"] }, tokio-tungstenite = { version = "0.24", features = ["rustls-tls-webpki-roots"] }, futures-util = "0.3", serde_json = "1"

use futures_util::{SinkExt, StreamExt};
use tokio_tungstenite::connect_async;
use tokio_tungstenite::tungstenite::{client::IntoClientRequest, Message};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut request = "wss://api.callouts.wtf/v1/stream".into_client_request()?;
    let key = std::env::var("CALLOUTS_KEY")?;
    request.headers_mut().insert("Authorization", format!("Bearer {key}").parse()?);
    let (mut ws, _) = connect_async(request).await?;
    ws.send(Message::Text(r#"{"type":"subscribe","all":false,"kinds":["original","update"]}"#.into())).await?;
    while let Some(message) = ws.next().await {
        if let Message::Text(text) = message? {
            let e: serde_json::Value = serde_json::from_str(&text)?;
            if e["type"] == "callout" {
                let ms = e["sent_at"].as_i64().unwrap_or(0) - e["pump_created_at"].as_i64().unwrap_or(0);
                println!("{} {} {ms} ms", e["caller"]["handle"], e["token"]["mint"]);
            }
        }
    }
    Ok(())
}
